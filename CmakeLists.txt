cmake_minimum_required(VERSION 3.22)
project(hobby_os LANGUAGES C CXX ASM_NASM)

set(ISO_DIR ${CMAKE_BINARY_DIR}/isodir)
set(GRUB_CFG ${CMAKE_SOURCE_DIR}/grub.cfg)

set(
    SOURCES
    src/multiboot2/header.asm
    src/multiboot2/main.asm
    src/multiboot2/main64.asm

    src/runtime/string.c
    src/hardware/vga/vga-text.cpp
    
    src/main.cpp
)

add_executable(hobby_os ${SOURCES})

target_include_directories(hobby_os PRIVATE include)

set_target_properties(hobby_os PROPERTIES 
    LINKER_LANGUAGE CXX
    LINK_DEPENDS ${CMAKE_SOURCE_DIR}/linker.ld
)

target_compile_options(hobby_os PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:-ffreestanding -fno-exceptions -fno-rtti -Wall -Wextra -g>
    $<$<COMPILE_LANGUAGE:C>:-ffreestanding -Wall -Wextra -g>
    $<$<COMPILE_LANGUAGE:ASM_NASM>:-g -F dwarf>
)

target_link_options(hobby_os PRIVATE
    -ffreestanding
    -nostdlib
    -T ${CMAKE_SOURCE_DIR}/linker.ld
)

if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows")
    set(GRUB_MKRESCUE_CMD wsl grub-mkrescue)
else()
    set(GRUB_MKRESCUE_CMD grub-mkrescue)
endif()

add_custom_command(
    OUTPUT ${CMAKE_BINARY_DIR}/hobby_os.iso
    COMMAND ${CMAKE_COMMAND} -E make_directory ${ISO_DIR}/boot/grub
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:hobby_os> ${ISO_DIR}/boot/hobby_os
    COMMAND ${CMAKE_COMMAND} -E copy ${GRUB_CFG} ${ISO_DIR}/boot/grub/grub.cfg
    COMMAND ${GRUB_MKRESCUE_CMD} -o hobby_os.iso isodir

    DEPENDS hobby_os ${GRUB_CFG}
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
)

add_custom_target(iso ALL DEPENDS ${CMAKE_BINARY_DIR}/hobby_os.iso)
